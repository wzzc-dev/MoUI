cmake_minimum_required(VERSION 3.22)
project(moui_embedding_harmonyos LANGUAGES C CXX)

if(NOT DEFINED MOUI_ROOT AND
   DEFINED ENV{MOUI_PACKAGE_ROOT} AND
   NOT "$ENV{MOUI_PACKAGE_ROOT}" STREQUAL "")
  set(MOUI_ROOT "$ENV{MOUI_PACKAGE_ROOT}")
endif()
if(NOT DEFINED MOUI_SHELL_ROOT AND
   DEFINED ENV{MOUI_SHELL_PACKAGE_ROOT} AND
   NOT "$ENV{MOUI_SHELL_PACKAGE_ROOT}" STREQUAL "")
  set(MOUI_SHELL_ROOT "$ENV{MOUI_SHELL_PACKAGE_ROOT}")
endif()
if(NOT DEFINED MOUI_ROOT OR NOT DEFINED MOUI_SHELL_ROOT)
  message(FATAL_ERROR "MOUI_ROOT and MOUI_SHELL_ROOT must point at matching published packages")
endif()

include("${MOUI_SHELL_ROOT}/harmonyos/embedder/cmake/MoUIShellHarmonyOS.cmake")
